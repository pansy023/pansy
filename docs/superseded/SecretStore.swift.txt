// MARK: - 密钥名字

/// `SecretStore` 里各条密钥的名字。
///
/// 沿用 Android 的 key 字面量（`password`，storage.md:93），这样：
/// 1) 迁移代码可以直接拿同一个字符串去 UserDefaults 里找遗留明文；
/// 2) 排查问题时「钥匙串条目名」与「原 SharedPreferences 键名」对得上。
public enum SecretKey {

    /// 教务系统密码。storage.md §2.1 :93 —— Android 端把它**明文**存在
    /// `timetable_prefs`，iOS 端改存钥匙串（`UserCredentials.swift:7-10` 的安全要求）。
    public static let jwxtPassword = "password"
}

// MARK: - 协议

/// 敏感字符串的落盘缝。
///
/// 为什么要有这个协议：`UserDefaultsTimetableRepository` 需要把密码写到**别处**去，
/// 而单元测试既不能碰真实钥匙串（macOS 上会弹权限框、需要签名才会成功），
/// 也不能碰真实用户偏好。把「密码放哪」抽成一个注入点，测试就能用
/// `InMemorySecretStore` 完全离线地跑。
///
/// 为什么方法带 `throws`（与任务描述里的示例签名略有出入）：
/// 钥匙串 API 会返回 `OSStatus` 失败码（例如设备锁定时 `errSecInteractionNotAllowed`），
/// 不抛错就只能吞掉 —— 那样密码会**静默写失败**，是比明文更糟的缺陷。
/// 抛出来之后由仓储层包成 `TimetableRepositoryError.writeFailed(key:reason:)`。
///
/// `AnyObject` 约束：实现是有状态的引用类型，避免值语义复制出两份「钥匙串」。
public protocol SecretStore: AnyObject {

    /// 读一条密钥；不存在返回 `nil`（**不要**用空串表示不存在，两者语义不同）。
    func string(for key: String) throws -> String?

    /// 写一条密钥；`nil` 表示**删除**该条目（对应 Android `putString(key, null)` 的语义）。
    func set(_ value: String?, for key: String) throws
}

// MARK: - 测试实现

/// 纯内存实现，给单元测试与预览用；绝不触碰钥匙串。
public final class InMemorySecretStore: SecretStore {

    /// 当前内容。测试可以直接断言密码到底进没进这里。
    public private(set) var storage: [String: String]

    /// 累计写入次数（含删除），便于测试确认「写过 / 没写过」。
    public private(set) var writeCount: Int = 0

    public init(storage: [String: String] = [:]) {
        self.storage = storage
    }

    public func string(for key: String) throws -> String? {
        storage[key]
    }

    public func set(_ value: String?, for key: String) throws {
        writeCount += 1
        if let value {
            storage[key] = value
        } else {
            storage.removeValue(forKey: key)
        }
    }
}

// MARK: - 生产实现

/// 钥匙串实现：一条 `kSecClassGenericPassword` 条目，`account` 就是 key。
///
/// 选型：generic password + `service` + `account` 是存「一组账号密码」的标准做法，
/// 不需要 `kSecAttrAccessGroup`（不跨 App 共享），也不需要 `kSecAttrSynchronizable`
/// （密码不上 iCloud，避免多设备串号）。
///
/// `kSecAttrAccessible` 取 `AfterFirstUnlock`：开机首次解锁后就一直可读。
/// 之所以不用 `WhenUnlocked`，是因为课程提醒要在后台（可能锁屏）读取凭据；
/// 之所以不用 `...ThisDeviceOnly`，是为了让用户换机恢复备份时不必重新登录。
public final class KeychainSecretStore: SecretStore {

    /// 钥匙串条目的 `kSecAttrService`。用包名保证与其它 App 隔离。
    public static let defaultService = "kebiaocom.example.kebiaozhu"

    /// 条目归属的服务名，测试可以注入别的值以便隔离。
    public let service: String

    public init(service: String = KeychainSecretStore.defaultService) {
        self.service = service
    }

    /// 读一条密钥。条目不存在返回 nil；`errSecItemNotFound` **不是**错误。
    public func string(for key: String) throws -> String? {
        #if canImport(Security)
        var query = baseQuery(for: key)
        query[kSecReturnData as String] = true
        query[kSecMatchLimit as String] = kSecMatchLimitOne
        var item: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &item)
        if status == errSecItemNotFound {
            return nil
        }
        guard status == errSecSuccess else {
            throw KeychainSecretStoreError.unexpectedStatus(status, operation: "SecItemCopyMatching")
        }
        guard let data = item as? Data else {
            throw KeychainSecretStoreError.unexpectedStatus(status, operation: "SecItemCopyMatching(非 Data)")
        }
        guard let value = String(data: data, encoding: .utf8) else {
            throw KeychainSecretStoreError.notUTF8(operation: "SecItemCopyMatching")
        }
        return value
        #else
        throw KeychainSecretStoreError.platformUnavailable(operation: "SecItemCopyMatching")
        #endif
    }

    /// 写一条密钥；`nil` 表示删除条目。
    ///
    /// 顺序是「先 `SecItemUpdate`，返回 `errSecItemNotFound` 再 `SecItemAdd`」：
    /// 钥匙串没有 upsert，直接 `Add` 已存在的条目会拿到 `errSecDuplicateItem`。
    public func set(_ value: String?, for key: String) throws {
        #if canImport(Security)
        let query = baseQuery(for: key)
        guard let value else {
            let status = SecItemDelete(query as CFDictionary)
            if status == errSecSuccess || status == errSecItemNotFound {
                return
            }
            throw KeychainSecretStoreError.unexpectedStatus(status, operation: "SecItemDelete")
        }
        let data = Data(value.utf8)
        let updates: [String: Any] = [kSecValueData as String: data]
        let updateStatus = SecItemUpdate(query as CFDictionary, updates as CFDictionary)
        if updateStatus == errSecSuccess {
            return
        }
        guard updateStatus == errSecItemNotFound else {
            throw KeychainSecretStoreError.unexpectedStatus(updateStatus, operation: "SecItemUpdate")
        }
        var insertion = query
        insertion[kSecValueData as String] = data
        insertion[kSecAttrAccessible as String] = kSecAttrAccessibleAfterFirstUnlock
        let addStatus = SecItemAdd(insertion as CFDictionary, nil)
        guard addStatus == errSecSuccess else {
            throw KeychainSecretStoreError.unexpectedStatus(addStatus, operation: "SecItemAdd")
        }
        #else
        throw KeychainSecretStoreError.platformUnavailable(operation: "SecItemAdd")
        #endif
    }

    #if canImport(Security)
    /// 三个条目属性：类型 generic password、服务名、账号名（= key）。
    ///
    /// `kSecUseDataProtectionKeychain` 让 macOS 也走 iOS 那套数据保护钥匙串
    /// （否则会落到 legacy 钥匙串，凭 ACL 授权、容易弹权限框）；iOS 上本就是它，写了也无害。
    private func baseQuery(for key: String) -> [String: Any] {
        var query: [String: Any] = [:]
        query[kSecClass as String] = kSecClassGenericPassword
        query[kSecAttrService as String] = service
        query[kSecAttrAccount as String] = key
        query[kSecUseDataProtectionKeychain as String] = true
        return query
    }
    #endif
}

// MARK: - 错误

/// 钥匙串层的错误。
///
/// `unexpectedStatus` 的关联值用 `Int32` 而**不是** `OSStatus`：`OSStatus` 是 Security
/// 框架的类型别名，用它就得让整个错误类型也依赖 Security；`Int32` 在任何平台都成立，
/// 而且数值与 `OSStatus` 完全一致（例如 `errSecDuplicateItem` = -25299）。
public enum KeychainSecretStoreError: Error, LocalizedError, Equatable {

    /// 钥匙串返回了非预期状态码。
    case unexpectedStatus(Int32, operation: String)

    /// 钥匙串里存的数据不是合法 UTF-8。
    case notUTF8(operation: String)

    /// 当前平台没有 Security 框架（非 Apple 平台编译时才会走到）。
    case platformUnavailable(operation: String)

    public var errorDescription: String? {
        switch self {
        case let .unexpectedStatus(status, operation):
            return "钥匙串操作 " + operation + " 失败，OSStatus " + String(status)
        case let .notUTF8(operation):
            return "钥匙串返回值不是合法 UTF-8：" + operation
        case let .platformUnavailable(operation):
            return "当前平台不支持钥匙串：" + operation
        }
    }
}
